Demo note: Bundle/import the SDK from your TypeScript application:

import { initVisualEditor } from "visual-editor-sdk";

initVisualEditor({
  siteId: "my-site",
  editorOrigin: "https://editor.example.com",
  fields: [
    { id: "home.hero.title", type: "text", label: "Hero Title" },
    {
      id: "home.articles",
      type: "data",
      label: "Articles Section",
      collection: "articles"
    },
    {
      id: "home.plans",
      type: "data",
      label: "Pricing Plans",
      collection: "plans"
    }
  ],
  onSelectItem(field, itemId, element) {
    console.log("Selected item:", itemId, "in", field.id);
  }
});
